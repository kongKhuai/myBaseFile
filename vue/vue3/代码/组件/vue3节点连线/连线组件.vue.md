<script setup>

import { computed, nextTick, onBeforeUnmount, onMounted, onUpdated, ref, watch } from "vue";

  

const props = defineProps({

  /**

   * 连线关系。每项：

   * from / to：相对本组件根节点的选择器，如 ".n1"

   * type："solid" 实线，"dashed" 虚线，默认虚线

   * color：描边颜色，默认 "#9ab8ab"

   * arrow："none" 无箭头，"both" 两侧，"left" 仅起点，"right" 仅终点，默认 "right"

   */

  edges: {

    type: Array,

    default: () => [],

  },

});

  

const rootRef = ref(null);

const lines = ref([]);

const uid = `node-link-${Math.random().toString(36).slice(2, 10)}`;

let resizeObserver = null;

  

const ARROW = { none: true, both: true, left: true, right: true };

  

function colorKey(color) {

  return encodeURIComponent(color).replace(/%/g, "");

}

  

function markerId(color) {

  return `${uid}-${colorKey(color)}`;

}

  

function normalize(edge) {

  if (!edge || !edge.from || !edge.to) return null;

  return {

    from: edge.from,

    to: edge.to,

    type: edge.type === "solid" ? "solid" : "dashed",

    arrow: ARROW[edge.arrow] ? edge.arrow : "right",

    color: edge.color || "#9ab8ab",

  };

}

  

const markers = computed(() => {

  const map = new Map();

  for (const edge of props.edges) {

    const item = normalize(edge);

    if (!item || item.arrow === "none") continue;

    map.set(item.color, { id: markerId(item.color), color: item.color });

  }

  return [...map.values()];

});

  

function find(root, selector) {

  try {

    return root.querySelector(selector);

  } catch {

    return null;

  }

}

  

function anchor(root, el, side) {

  const nodeBox = el.getBoundingClientRect();

  const rootBox = root.getBoundingClientRect();

  const x = side === "right" ? nodeBox.right - rootBox.left : nodeBox.left - rootBox.left;

  const y = nodeBox.top + nodeBox.height / 2 - rootBox.top;

  return {

    x: Math.round(x * 10) / 10,

    y: Math.round(y * 10) / 10,

  };

}

  

function buildLines() {

  const root = rootRef.value;

  if (!root) return [];

  const next = [];

  props.edges.forEach((edge, index) => {

    const item = normalize(edge);

    if (!item) return;

    const fromEl = find(root, item.from);

    const toEl = find(root, item.to);

    if (!fromEl || !toEl) return;

    const start = anchor(root, fromEl, "right");

    const end = anchor(root, toEl, "left");

    const marker = `url(#${markerId(item.color)})`;

    next.push({

      key: `${item.from}>${item.to}:${index}`,

      x1: start.x,

      y1: start.y,

      x2: end.x,

      y2: end.y,

      color: item.color,

      dash: item.type === "dashed" ? "5 4" : null,

      markerStart: item.arrow === "left" || item.arrow === "both" ? marker : null,

      markerEnd: item.arrow === "right" || item.arrow === "both" ? marker : null,

    });

  });

  return next;

}

  

function sameLines(current, next) {

  if (current.length !== next.length) return false;

  return current.every((line, index) => {

    const other = next[index];

    return (

      line.x1 === other.x1 &&

      line.y1 === other.y1 &&

      line.x2 === other.x2 &&

      line.y2 === other.y2 &&

      line.color === other.color &&

      line.dash === other.dash &&

      line.markerStart === other.markerStart &&

      line.markerEnd === other.markerEnd

    );

  });

}

  

function redraw() {

  const next = buildLines();

  if (sameLines(lines.value, next)) return;

  lines.value = next;

}

  

function scheduleRedraw() {

  nextTick(() => {

    requestAnimationFrame(redraw);

  });

}

  

watch(() => props.edges, scheduleRedraw, { deep: true });

  

onMounted(() => {

  resizeObserver = new ResizeObserver(redraw);

  resizeObserver.observe(rootRef.value);

  window.addEventListener("resize", redraw);

  scheduleRedraw();

});

  

onUpdated(redraw);

  

onBeforeUnmount(() => {

  resizeObserver?.disconnect();

  window.removeEventListener("resize", redraw);

});

  

defineExpose({ redraw });

</script>

  

<template>

  <div ref="rootRef" class="node-link">

    <svg class="node-link__svg" aria-hidden="true">

      <defs>

        <marker v-for="marker in markers" :id="marker.id" :key="marker.id" viewBox="0 0 10 10" refX="9" refY="5"

          markerWidth="6" markerHeight="6" orient="auto-start-reverse">

          <path d="M0,0 L10,5 L0,10 z" :fill="marker.color" />

        </marker>

      </defs>

      <line v-for="line in lines" :key="line.key" :x1="line.x1" :y1="line.y1" :x2="line.x2" :y2="line.y2"

        :stroke="line.color" stroke-width="1.2" :stroke-dasharray="line.dash" :marker-start="line.markerStart"

        :marker-end="line.markerEnd" />

    </svg>

    <slot />

  </div>

</template>

  

<style scoped>

.node-link {

  position: relative;

}

  

.node-link__svg {

  position: absolute;

  inset: 0;

  width: 100%;

  height: 100%;

  overflow: visible;

  pointer-events: none;

}

</style>
