			let angleValue = angle;

		/**
		 * 将 Base64 图片数据转换为本地临时文件路径
		 * @param {string} base64 - 完整的 base64 字符串 (含 data:image/xxx;base64, 前缀)
		 * @param {Function} callback - (tempFilePath, error) => {}
		 */
		base64ToTempFile(base64, callback) {
			// 1. 分离纯 base64
			const base64Arr = base64.split(',');
			if (base64Arr.length < 2) {
				callback(null, new Error('Base64 数据无效'));
				return;
			}
			const [prefix, base64Data] = base64Arr;

			// 区分环境：H5 / App/小程序
			// #ifdef H5
			// H5 浏览器环境：转 Blob + 临时 URL
			const mime = prefix.match(/data:(.+?);/)?.[1] || 'image/png';
			const bstr = atob(base64Data);
			let n = bstr.length;
			const u8arr = new Uint8Array(n);
			while (n--) {
				u8arr[n] = bstr.charCodeAt(n);
			}
			const blob = new Blob([u8arr], { type: mime });
			const tempUrl = URL.createObjectURL(blob);
			callback(tempUrl, null);
			// #endif

			// #ifndef H5
			// App / 小程序 原有逻辑
			const fileName = `watermark_${Date.now()}_${Math.random().toString(36).substr(2, 8)}.png`;
			const filePath = `${uni.env.USER_DATA_PATH}/${fileName}`;

			// 写入本地文件
			uni.writeFile({
				filePath,
				data: base64Data,
				encoding: 'base64',
				success: () => {
					callback(filePath, null);
				},
				fail: (err) => {
					callback(null, err);
				}
			});
			// #endif
