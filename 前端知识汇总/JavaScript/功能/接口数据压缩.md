import pako from "pako"



/**
 * 判断入参是否为 JSON 字符串
 * - 是 → 直接 JSON.parse 返回对象
 * - 否 → 调用函数2处理（解码+解压+解析）
 */
function parseOrProcess(input) {
	if (typeof input !== 'string') {
		throw new TypeError('解压数据函数输入必须是字符串格式，当前是', typeof input);
	}
	try {
		const parsed = JSON.parse(input);
		return parsed; // 是 JSON，直接返回
	} catch (_) {
		// 不是 JSON，交给函数2
		return decodeBase64AndGunzip(input);
	}
}

/**
 * 函数2：Base64 解码 → Gzip 解压 → JSON 解析
 * 若任何步骤失败，返回原始 base64Str
 * 依赖全局 pako (Gzip 解压)
 */
function decodeBase64AndGunzip(base64Str) {
	try {
		// 1. Base64 解码为 Uint8Array
		const binary = atob(base64Str);
		const bytes = new Uint8Array(binary.length);
		for (let i = 0; i < binary.length; i++) {
			bytes[i] = binary.charCodeAt(i);
		}

		// 2. Gzip 解压
		const inflated = pako.ungzip(bytes, { to: 'string' });

		// 3. 解析 JSON
		return JSON.parse(inflated);
	} catch (error) {
		// 任何失败（解码、解压、解析）返回原始数据
		return base64Str;
	}
}
