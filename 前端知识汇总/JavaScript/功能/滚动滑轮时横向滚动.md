 //滚动滑轮时横向滚动

  const scoolXBox = document.querySelector(".scoolXBox");

  if (!scoolXBox) return;

  scoolXBox.addEventListener("wheel", (event) => {

    event.preventDefault();

    scoolXBox.scrollBy({

      left: event.deltaY, // 将纵向滚动的值映射到横向滚动

      behavior: "smooth", // 使滚动更平滑

    });

  });
