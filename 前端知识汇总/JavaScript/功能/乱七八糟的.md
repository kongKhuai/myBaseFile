/**  
 * 任务事件根据类型判断返回iso格式的gantt可以解析的时间  
 * @param time 任务时间  根据时间小时数区分上午还是下午  
 * @param type 开始（start）/结束 （end）     
* @returns {*} isoTime  
 */function getReskTime2Iso(time,type='start'){  
    const timer = moment(time);  
    const isMorning = timer.hour() < 12;//是否为上午  
    let hour = 0;  
    let min = 0  
    // 开始的  
    if(type=='start'){  
        hour = isMorning ? 0 : 12;  
        min = isMorning ? 0 : 1;  
    }else{//结束的  
        hour = isMorning ? 11 : 23;  
        min = 59  
    }  
    // .format('YYYY-MM-DD HH:mm');  
    return  timer  
            .hour(hour)  
            .minute(min).toISOString();  
}

/**  
 * 判断目标日期字符串是否在 lastDate 之前3天内（含边界）  
 * @param {string} contrast - contrast 对比的日期字符串 或者Date类型  
 * @param {string} targetStr - 待判断的日期字符串 或者Date类型  
 * @returns {boolean} 是否在范围内  
 */  
isTargetWithinDaysBefore(contrast, targetStr, day = 2) {  
    if (typeof day !== "number") return false;  
    const originalDate = new Date(contrast);  
    const lastDate = originalDate;  
    lastDate.setHours(23, 59, 59, 999);  
  
    const targetDate = new Date(targetStr);  
    targetDate.setHours(23, 59, 59, 999);  
  
    const threeDaysMs = day * 24 * 60 * 60 * 1000;  
    const timeDiff = lastDate - targetDate;  
    return timeDiff >= 0 && timeDiff <= threeDaysMs;  
},
